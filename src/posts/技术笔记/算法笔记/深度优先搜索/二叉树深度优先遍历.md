#### (1) 前序遍历
```javascript
var preorderTraversal = function(root) {
    const res = [];
    
    const traverse = (node) => {
        if (!node) return;
        
        res.push(node.val);  // 1. 访问根节点
        traverse(node.left); // 2. 递归左子树
        traverse(node.right);// 3. 递归右子树
    };
    
    traverse(root);
    return res;
};
```
