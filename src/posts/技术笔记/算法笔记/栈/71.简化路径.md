/**

* @param {string} path

* @return {string}

*/

var simplifyPath = function(path) {

	const arr = path.split('/').filter(item => !!item)
	
	const result = []
	
	for(let p of arr){
	
		if(p === '..'){
		
			result.pop()
		
		}else if(p === '.'){ // 如果是.不需要处理
		
			continue;
		
		}else{
		
			result.push(p)
		
		}
	
	}
	
	return '/'+result.filter(item => !!item).join('/')

};
