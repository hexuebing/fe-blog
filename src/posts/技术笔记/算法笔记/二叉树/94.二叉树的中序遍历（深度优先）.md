var inOrderTraversal = function(root) {
    let result = []
    // 定义一个遍历函数，用来将遍历的数放到结果队列中
    function travel(curNode){
        // 如果节点不存在直接返回
        if(curNode === null) return
        if(curNode.left)  travel(curNode.left)
        result.push(curNode.val)
        if(curNode.right) travel(curNode.right)
    }
    travel(root)
    return result
}
