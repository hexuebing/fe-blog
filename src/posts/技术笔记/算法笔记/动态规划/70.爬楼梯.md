/**
 * @param {number} n
 * @return {number}
 */
var climbStairs = function(n) {
    // 处理特殊情况
    if (n <= 2) return n;

    // 初始化 dp 数组
    const dp = new Array(n + 1);
    dp[1] = 1;
    dp[2] = 2;

    // 从第 3 阶开始计算
    for (let i = 3; i <= n; i++) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }

    return dp[n];
};
