var groupAnagrams = function(strs) {
    const map = new Map();
    
    for (let str of strs) {
        // 将字符串排序作为 key
        const key = str.split('').sort().join('');
        
        // 如果 key 不存在，创建新数组；否则添加到现有数组
        if (!map.has(key)) {
            map.set(key, [str]);
        } else {
            map.get(key).push(str);
        }
    }
    
    // 将 Map 的所有值转换为数组
    return Array.from(map.values());
};
